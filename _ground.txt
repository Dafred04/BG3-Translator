@dataclass
class AppSettings:
    """Application settings."""
    # General
    language: str = "en"
    theme: str = "system"  # light, dark, system
    auto_save: bool = True
    auto_save_interval: int = 30  # seconds

    # Translation
    default_provider: str = "gemini"
    providers: Dict[str, TranslationProviderConfig] = field(default_factory=dict)
    batch_size: int = 50
    max_concurrent_requests: int = 3
    retry_attempts: int = 3
    timeout: int = 60

    # Dictionary/Cache
    use_dictionary: bool = True
    dictionary_match_threshold: float = 0.85
    cache_translations: bool = True

    # Validation
    validate_placeholders: bool = True
    validate_terminology: bool = True
    strict_mode: bool = False

    # Export
    export_format: str = "pak"  # pak, xml, zip
    include_original: bool = False
    create_backup: bool = True

    # Advanced
    log_level: str = "INFO"
    debug_mode: bool = False

    @classmethod
    def load(cls, path: Path) -> "AppSettings":
        if path.exists():
            try:
                with open(path, "r", encoding="utf-8") as f:
                    data = json.load(f)
                settings = cls()
                for key, value in data.items():
                    if hasattr(settings, key):
                        if key == "providers":
                            settings.providers = {
                                k: TranslationProviderConfig(**v) for k, v in value.items()
                            }
                        else:
                            setattr(settings, key, value)
                return settings
            except Exception:
                pass
        return cls()

    def save(self, path: Path):
        data = {}
        for key in self.__dataclass_fields__:
            value = getattr(self, key)
            if key == "providers":
                data[key] = {k: v.__dict__ for k, v in value.items()}
            else:
                data[key] = value
        with open(path, "w", encoding="utf-8") as f:
            json.dump(data, f, indent=2, ensure_ascii=False)

class Toast(QFrame):
    """Floating notification, auto-dismisses."""

    CLOSED = Signal()

    def __init__(self, message: str, kind: str = "info", timeout_ms: int = 3500, parent=None):
        super().__init__(parent)
        self.setObjectName("Toast")
        if kind == "success":
            self.setProperty("class", "ToastSuccess")
            self.setObjectName("ToastSuccess")
        elif kind == "error":
            self.setObjectName("ToastError")
        elif kind == "warning":
            self.setObjectName("ToastWarning")

        icons = {"info": "ℹ", "success": "✔", "error": "✖", "warning": "⚠"}
        lay = QHBoxLayout(self)
        icon = QLabel(icons.get(kind, "ℹ"))
        icon.setFont(QFont("Segoe UI Emoji", 14))
        msg = QLabel(message)
        msg.setWordWrap(True)
        lay.addWidget(icon)
        lay.addWidget(msg, 1)

        self.setFixedWidth(380)
        self.setMinimumHeight(48)

        shadow = QGraphicsDropShadowEffect(blurRadius=24, xOffset=0, yOffset=6)
        shadow.setColor(self.palette().color(self.palette().ColorRole.Shadow))
        self.setGraphicsEffect(shadow)

        self.timer = QTimer(self)
        self.timer.setSingleShot(True)
        self.timer.timeout.connect(self._close)
        self.timer.start(timeout_ms)

    def _close(self):
        self.CLOSED.emit()
        self.deleteLater()

    def set_timeout(self, ms: int):
        self.timer.start(ms)

"""Shared application state (current project, services)."""

from typing import Optional

from PySide6.QtCore import QObject, Signal

from ..config import PATHS, get_settings
from ..models.translation import TranslationEntry, TranslationProject, ModMetadata
from ..services.dictionary import DictionaryStore
from ..services.engine import TranslationEngine
from ..services.providers import get_provider


class AppState(QObject):
    """Application-wide state, shared between pages via MainWindow."""

    project_changed = Signal()
    entry_edited = Signal(str)  # handle

    def __init__(self):
        super().__init__()
        self.settings = get_settings()
        self.dictionary = DictionaryStore(PATHS.cache_dir / "dictionary.db")
        self.project: Optional[TranslationProject] = None
        self._engine: Optional[TranslationEngine] = None

    def _build_provider(self):
        prov_cfg = self.settings.providers.get(self.settings.default_provider)
        try:
            return get_provider(
                self.settings.default_provider,
                api_key=prov_cfg.api_key if prov_cfg else "",
                model=prov_cfg.model if prov_cfg else "",
                base_url=prov_cfg.base_url if prov_cfg else "",
                timeout=self.settings.timeout,
            )
        except Exception:
            return None

    @property
    def engine(self) -> TranslationEngine:
        if self._engine is None:
            self._engine = TranslationEngine(self.dictionary, provider=self._build_provider())
        return self._engine

    def reset_engine(self):
        """Rebuild the provider in-place, keeping the same engine object so UI
        signal connections stay valid (avoids disconnect warnings)."""
        if self._engine is None:
            self._engine = TranslationEngine(self.dictionary, provider=self._build_provider())
        else:
            self._engine.set_provider(self._build_provider())

    def new_project(self, metadata: ModMetadata, entries: list[TranslationEntry],
                    source_lang: str, target_lang: str):
        self.project = TranslationProject(
            metadata=metadata,
            entries=entries,
            source_language=source_lang,
            target_language=target_lang,
        )
        self.project_changed.emit()

    def close_project(self):
        self.project = None
        self.project_changed.emit()

    def stats(self) -> dict:
        if not self.project:
            return {"total": 0, "translated": 0, "percent": 0, "errors": 0}
        p = self.project
        return {
            "total": p.total_entries,
            "translated": p.translated_entries,
            "percent": int(p.get_progress() * 100),
            "errors": sum(1 for e in p.entries if e.has_validation_errors()),
        }


_app_state: Optional[AppState] = None


def get_app_state() -> AppState:
    global _app_state
    if _app_state is None:

class ImportWorker(QObject):
    """Background import: reads .xml/.pak/.zip/folder, returns a project.

    Note: full .pak (LSPK v18) decompression is handled by core.pak (M6).
    This worker supports .xml and .zip (containing .xml) and folders now;
    .pak falls back to scanning for nested XMLs after extraction attempt.
    """

    finished = Signal(object)   # TranslationProject
    failed = Signal(str)

    def __init__(self, path: str, is_folder: bool, source_lang: str, target_lang: str):
        super().__init__()
        self.path = path
        self.is_folder = is_folder
        self.source_lang = source_lang
        self.target_lang = target_lang

    def run(self):
        try:
            project = self._import()
            self.finished.emit(project)
        except Exception as exc:
            self.failed.emit(str(exc))

    def _import(self) -> TranslationProject:
        import tempfile
        p = Path(self.path)
        xml_files: list[Path] = []
        temp_dir = None

        if self.is_folder or p.is_dir():
            xml_files = self._find_xmls(p)
        elif p.suffix.lower() == ".xml":
            xml_files = [p]
        elif p.suffix.lower() == ".zip":
            temp_dir = Path(tempfile.mkdtemp(prefix="bg3mt_"))
            with zipfile.ZipFile(p, "r") as zf:
                zf.extractall(temp_dir)
            xml_files = self._find_xmls(temp_dir)
        elif p.suffix.lower() == ".pak":
            from ...core.pak import extract_pak, PakError
            temp_dir = Path(tempfile.mkdtemp(prefix="bg3mt_pak_"))
            try:
                extract_pak(p, temp_dir)
            except PakError as exc:
                raise RuntimeError(f"Не удалось прочитать пакет: {exc}") from exc
            xml_files = self._find_xmls(temp_dir)
        else:
            raise RuntimeError(f"Неподдерживаемый тип файла: {p.suffix}")

        if not xml_files:
            raise RuntimeError("В источнике не найдено файлов локализации (XML)")

        entries = []
        for xf in xml_files:
            text = xf.read_text(encoding="utf-8", errors="replace")
            parsed = parse_loca_xml(text)
            entries.extend(loca_to_entries(parsed, self.source_lang, self.target_lang))

        if not entries:
            raise RuntimeError("Файлы локализации не содержат строк")

        metadata = ModMetadata(
            name=p.stem,
            source_path=p,
            supported_languages=[self.target_lang],
        )
        return TranslationProject(
            metadata=metadata,
            entries=entries,
            source_language=self.source_lang,
            target_language=self.target_lang,
        )

    @staticmethod
    def _find_xmls(root: Path) -> list[Path]:
        result = []
        for pattern in ("**/*.xml", "**/*.loc"):
            result.extend(root.glob(pattern))
        # Prefer files under Localization folders first
        result.sort(key=lambda x: (0 if "localization" in str(x).lower() else 1, str(x)))
        return result

    def _apply_filter(self):

        by_handle = {e.handle: e for e in project.entries}

            entry = by_handle.get(handle)
            match_text = (not needle) or (needle in src) or (needle in tgt)
            match_mode = True
            if mode == "untranslated":
                match_mode = bool(entry) and not entry.is_translated()
            elif mode == "translated":
                match_mode = bool(entry) and entry.is_translated()
            elif mode == "errors":
                match_mode = bool(entry and entry.has_validation_errors())
            self.table.setRowHidden(row, not (match_text and match_mode))

    def _translate_all(self):

            return

        # once in __init__, so nothing to wire up here.
        engine = self.state.engine

        self.progress.setValue(0)
        self.translate_btn.setEnabled(False)
        self.cancel_btn.setVisible(True)
        self.cancel_btn.setEnabled(True)

        # Wrap only the untranslated entries; the engine mutates them in place,
        # so the live project objects are updated directly.
        sub = TranslationProject(
            metadata=project.metadata,
            entries=list(targets),
            source_language=project.source_language,

def _fmt_time(seconds: float) -> str:
    """Format a duration in a compact human form."""
    if seconds <= 0:
        return "—"
    seconds = int(seconds)
    h, rem = divmod(seconds, 3600)
    m, s = divmod(rem, 60)
    if h:
        return f"{h} ч {m:02d} мин"
    if m:
        return f"{m} мин {s:02d} с"
    return f"{s} с"
